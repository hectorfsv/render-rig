<script>
(function(){
  /* THE PHONE CONSOLE, pass 2 (2026-09-26): THE PHONE IS A FORM. Hector on the live
     page: "on iphone everything's so tight together and hard to understand." In portrait,
     or any tall window, the console is one column: the modes wrapped so all eight are in
     view, the stage a strip until something renders, then the composer scrolls as a form
     (prompt, every setting as a labelled section, Sources, Session) with the price and
     Generate pinned under it. A phone on its side keeps the 2026-09-23 sheet: those
     checks are the `else` branch, unchanged. What both must guarantee: Generate always
     reachable, the page never scrolls, the chosen mode's tab pressed and in view. */
  var S=String.fromCharCode(167), out=[], q=new URLSearchParams(location.search);
  window.fetch=function(u,o){var b=o&&o.body,a=(b&&b.get)?b.get('action'):'';
    // as the owner since 2026-09-24: Talk is the owner's mode
    if(a==='auth')return Promise.resolve({ok:true,status:200,json:function(){return Promise.resolve({guest:false,user:'hector'})},text:function(){return Promise.resolve('')}});
    return Promise.reject(new Error('no net'));};
  function $(i){return document.getElementById(i)}
  function vis(el){ if(!el) return false; var r=el.getBoundingClientRect();
    return r.width>0&&r.height>0&&getComputedStyle(el).display!=='none'; }
  function onScreen(el){ var r=el.getBoundingClientRect();
    return r.top>=-1 && r.bottom<=innerHeight+1 && r.width>0; }
  function hit(el){var r=el.getBoundingClientRect();
    if(r.width<1||r.height<1)return 'zero';
    if(r.top<0||r.bottom>innerHeight)return 'offscreen';
    var e=document.elementFromPoint(r.left+r.width/2,r.top+r.height/2);
    return e&&(el.contains(e)||e.contains(el))?'yes':'NO';}
  function T(c,msg){ out.push((c?'PASS':'FAIL')+'  '+msg); }
  function chip(id){ return document.querySelector('#chips .chip-s[data-row="'+id+'"]') }
  function openRows(){ return [].filter.call($('pane-ctl').querySelectorAll('.ctl-scroll > *'),vis) }
  function box(){ var r=$('pane-stage').getBoundingClientRect(); return r }
  var form = innerHeight>innerWidth || innerHeight>=561;   // the same rule as the page's formMode()
  function go(){
    $('pw').value='x'; $('unlock').click();
    setTimeout(function(){
      // THE HUB, before we leave it. It was never measured on a phone, which is
      // how a login screen that pans the whole rig off its own left edge shipped.
      var d0=document.documentElement, rail=document.querySelector('.rail'),
          rr=rail.getBoundingClientRect(), cards=[].slice.call(document.querySelectorAll('.card'));
      T(d0.scrollWidth<=innerWidth, 'hub: no horizontal overflow ('+d0.scrollWidth+'<='+innerWidth+')');
      T(rr.left>=0 && rr.right<=innerWidth+1 && rr.width>0,
        'hub: the rail is on screen (x '+Math.round(rr.left)+'-'+Math.round(rr.right)+' of '+innerWidth+')');
      // Derived, not hard-coded: the hub must carry exactly one card per mode.
      var modes=document.querySelectorAll('#seg-type button').length;
      T(cards.length===modes, 'hub: one card per mode ('+cards.length+' cards, '+modes+' modes)');
      T(cards.every(function(c){var r=c.getBoundingClientRect();return r.left>=0&&r.right<=innerWidth+1}),
        'hub: no card is clipped sideways');
      var m=q.get('m')||'image';
      document.querySelector('.card[data-go="'+m+'"]').click();
      setTimeout(function(){
        var d=document.documentElement;
        T(d.scrollWidth<=innerWidth, 'no horizontal overflow ('+d.scrollWidth+'<='+innerWidth+')');
        var mk=document.querySelector('.mark');
        T(mk.scrollWidth<=Math.ceil(mk.getBoundingClientRect().width),
          'header title does not paint over the buttons (needs '+mk.scrollWidth+
          ', has '+Math.round(mk.getBoundingClientRect().width)+')');
        T(d.scrollHeight<=innerHeight+1, 'console does not scroll the page ('+d.scrollHeight+' vs '+innerHeight+')');
        T(onScreen($('submit')), 'Generate is on screen');
        T(hit($('submit'))==='yes', 'Generate is tappable ('+hit($('submit'))+')');
        T(onScreen($('cost')), 'cost meter is on screen');
        T(vis($('pane-stage')), 'stage is visible');
        var noWords=(m==='upscale'||m==='camera'||m==='talk');
        T(noWords ? !vis($('prompt')) : (vis($('prompt')) && onScreen($('prompt'))),
          noWords ? m+' shows no prompt (it needs none)' : 'the prompt is on screen');
        // the chosen mode's tab is pressed and fully inside the strip
        var strip=$('seg-type'), tb=strip.querySelector('button[aria-pressed=true]'),
            sr=strip.getBoundingClientRect(), br=tb?tb.getBoundingClientRect():null;
        T(tb && tb.dataset.v===m, 'the '+m+' tab is the pressed one');
        T(br && br.left>=sr.left-1 && br.right<=sr.right+1, 'the '+m+' tab is in view in the mode strip'+
          (br ? ' (tab '+Math.round(br.left)+'-'+Math.round(br.right)+', strip '+Math.round(sr.left)+'-'+Math.round(sr.right)+
                ', scroll '+strip.scrollLeft+' of '+(strip.scrollWidth-strip.clientWidth)+')' : ''));
        var st0=box();
        out.push('      stage at open '+Math.round(st0.width)+'x'+Math.round(st0.height)+
                 '  = '+Math.round(st0.height/innerHeight*100)+'% of the screen');

        if(form){
          /* ---------- the form ---------- */
          T(!vis($('chips')), 'the form has no chip row');
          T(strip.scrollWidth<=strip.clientWidth+1, 'all '+modes+' modes are in view, nothing scrolls sideways ('+strip.scrollWidth+' of '+strip.clientWidth+')');
          T(st0.height<=90, 'the empty stage is a strip, not the screen ('+Math.round(st0.height)+'px)');
          if(!noWords) T($('prompt').getBoundingClientRect().height>=80, 'the prompt shows three lines ('+Math.round($('prompt').getBoundingClientRect().height)+'px)');
          var rows=openRows();
          T(rows.length>=1, 'the settings are sections of the form ('+rows.map(function(r){return r.id}).join(' ')+')');
          T(rows.every(function(r){ return r.dataset.name || r.querySelector('.lab, summary, .check, .hint') }),
            'every section carries a heading');
          T(vis($('pane-src')) && vis(document.querySelector('.hist')), 'Sources and Session are sections of the form');
          // every section scrolls into view inside the composer, and Generate never moves
          var comp=$('composer'), gy=$('submit').getBoundingClientRect().top, bad=[];
          rows.concat([$('pane-src'), document.querySelector('.hist')]).forEach(function(r){
            r.scrollIntoView({block:'start'});
            var rr=r.getBoundingClientRect();
            if(!(rr.top>=-1 && rr.top<innerHeight) || !onScreen($('submit'))) bad.push(r.id||r.className);
          });
          T(!bad.length, 'every section scrolls into view with Generate still on screen'+(bad.length?' | BAD '+bad.join(' '):''));
          T(Math.abs($('submit').getBoundingClientRect().top-gy)<1, 'Generate does not move while the form scrolls');
          T(comp.scrollHeight>comp.clientHeight ? comp.scrollTop>0 : true, 'the form scrolls inside the composer ('+(comp.scrollHeight-comp.clientHeight)+'px of it)');
          T(d.scrollHeight<=innerHeight+1, 'still no page scroll after scrolling the form');
          T(hit($('submit'))==='yes', 'Generate is still tappable over the scrolled form ('+hit($('submit'))+')');
          // a render arriving gives the stage the screen back
          $('stage-idle').hidden=true; $('stage-out').hidden=false;
          var st2=box();
          T(st2.height>=100, 'with a render the stage gets real room ('+Math.round(st2.height)+'px)');
          $('stage-idle').hidden=false; $('stage-out').hidden=true;
          T(box().height<=90, 'and folds back to a strip when cleared ('+Math.round(box().height)+'px)');
        } else {
          /* ---------- a phone on its side: the sheet, unchanged since 2026-09-23 ---------- */
          T(vis($('chips')) && $('chips').children.length>0, 'the settings chips are on screen ('+$('chips').children.length+')');
          T(!vis($('pane-ctl')) && !vis($('pane-src')) && !vis(document.querySelector('.hist')),
            'the console opens stage-first (no panel open)');
          var ids=[].map.call(document.querySelectorAll('#chips .chip-s:not(.ph)'),function(c){return c.dataset.row});
          var bad2=[];
          ids.forEach(function(id){
            chip(id).click();
            var rows2=openRows(), ok=vis($('pane-ctl')) && rows2.length===1 && rows2[0].id===id
                     && chip(id).getAttribute('aria-pressed')==='true' && onScreen($('submit'));
            if(!ok) bad2.push(id+'('+rows2.map(function(r){return r.id}).join('+')+')');
          });
          T(ids.length>0 && !bad2.length, 'every setting chip opens exactly its own setting, Generate on screen ('+
            ids.join(' ')+(bad2.length?' | BAD '+bad2.join(' '):'')+')');
          var st1=box();
          out.push('      stage with a setting open '+Math.round(st1.width)+'x'+Math.round(st1.height));
          T(st1.height>=100 || st1.width>=240, 'the stage keeps real room with a setting open ('+Math.round(st1.width)+'x'+Math.round(st1.height)+')');
          T(d.scrollHeight<=innerHeight+1, 'no page scroll with a setting open');
          var pc=$('pane-ctl');
          T(pc.clientHeight >= Math.min(120, pc.scrollHeight-2), 'the open setting shows at least 120px of itself ('+pc.clientHeight+' of '+pc.scrollHeight+')');
          T(!vis($('pane-src')) && !vis(document.querySelector('.hist')), 'exactly one panel open (the setting)');
          var p=$('pane-ctl'), gy2=$('submit').getBoundingClientRect().top; p.scrollTop=p.scrollHeight;
          T(p.scrollHeight>p.clientHeight ? p.scrollTop>0 : true,
            'the open setting scrolls internally ('+(p.scrollHeight-p.clientHeight)+'px)');
          T(Math.abs($('submit').getBoundingClientRect().top-gy2)<1 && onScreen($('submit')), 'Generate does not move while the panel scrolls');
          chip(ids[ids.length-1]).click();
          var st3=box();
          T(!vis($('pane-ctl')), 'tapping the open chip closes the setting');
          T(st3.width*st3.height>=st1.width*st1.height, 'closing gives the stage its room back ('+
            Math.round(st1.width)+'x'+Math.round(st1.height)+' -> '+Math.round(st3.width)+'x'+Math.round(st3.height)+')');
          chip('pane-src').click();
          T(vis($('pane-src')) && !vis($('pane-ctl')) && !vis(document.querySelector('.hist')), 'the Sources chip opens Sources alone');
          T(onScreen($('submit')), 'Generate still on screen with Sources open');
          chip('hist').click();
          T(vis(document.querySelector('.hist')) && !vis($('pane-src')), 'the Session chip swaps the panel');
          chip('hist').click();
          T(!vis(document.querySelector('.hist')) && !vis($('pane-src')) && !vis($('pane-ctl')), 'tapping the open Session chip closes the sheet');
          T(onScreen($('submit')), 'Generate survives the collapse');
          T(d.scrollHeight<=innerHeight+1, 'still no page scroll when collapsed');
        }
        document.title=S+out.join('\n')+S;
      },200);
    },150);
  }
  if(document.readyState==='complete')go();else addEventListener('load',go);
})();
</script>
