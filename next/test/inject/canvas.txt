  function staleness(){
    /* ---------- fresh, stale and never paying twice ---------- */
    R.importBoard({v:1,name:'',nodes:[
      {id:'n1',kind:'gallery',x:0,y:0,src:{url:'https://v3.fal.media/files/g.png',w:2048,h:1152}},
      {id:'n2',kind:'compose',x:320,y:0,settings:{prompt:'a'}},
      {id:'n3',kind:'lite',x:660,y:0,settings:{prompt:'b'}},
      {id:'n4',kind:'image',x:0,y:420,settings:{prompt:'c'}}],
      edges:[['n1:out','n2:images'],['n2:out','n3:image']]});
    eq(R.totals().run,['n2','n3','n4'],'a new board: every engine is to run');
    eq(R.totals().sum,0.61,'…$0.15 + $0.40 + $0.06 = $0.61');
    ok(/\$0\.61/.test($('total').textContent)&&/3 to run/.test($('total').textContent),'…the header says so',$('total').textContent);
    R.setRan('n2',{sig:R.info('n2').sig,url:png(64,36,'#c84'),w:2048,h:1152,price:0.15,job:1});
    eq(st('n2'),'fresh','a node that ran with these settings is fresh');
    ok(card('n2').querySelector('.pic img'),'…and shows its result');
    R.setRan('n3',{sig:R.info('n3').sig,url:'https://v3.fal.media/files/v.mp4',price:0.4,job:2});
    R.setRan('n4',{sig:R.info('n4').sig,url:png(32,32,'#48c'),w:1024,h:1024,price:0.06,job:3});
    eq([st('n2'),st('n3'),st('n4')],['fresh','fresh','fresh'],'all three ran: all fresh');
    eq([R.totals().sum,$('total').textContent],[0,'nothing to pay'],'nothing to pay: a fresh node is never paid for twice');
    R.set('n2','prompt','a2');
    eq([st('n2'),st('n3'),st('n4')],['changed','changed','fresh'],'change Compose: it and what it feeds go stale, the rest stays fresh');
    eq([R.totals().run,R.totals().sum],[['n2','n3'],0.55],'…$0.55 to pay, for those two only');

    R.set('n3','dur','6'); eq(st('n3'),'changed','a setting that is sent does'); R.undo();
    /* ---------- freeze ---------- */
    R.set('n2','prompt','a3');
    R.freeze('n2',true);
    eq([st('n2'),st('n3')],['frozen','fresh'],'freeze a changed Compose: it keeps its result, and what it feeds stays fresh');

    ok(card('n2').querySelector('fieldset').disabled,'…its settings are locked');
    R.set('n2','prompt','zzz'); eq(node('n2').settings.prompt,'a3','…and cannot change');
    eq(card('n2').querySelector('[data-act="freeze"]').getAttribute('aria-pressed'),'true','…the snowflake shows it');
    R.freeze('n2',false); eq([st('n2'),st('n3')],['changed','changed'],'unfreeze: stale again');
    var k5=R.add('image').id; R.set(k5,'prompt','p'); var u5=R.add('upscale').id; R.connect(k5+':out',u5+':image');

    ok(R.totals().needs.indexOf(k5)>=0,'…and it is counted as needing something');
    R.freeze(k5,false); R.remove(u5); R.remove(k5);
    ok(!card('n1').querySelector('[data-act="freeze"]'),'a source has nothing to freeze');
  }

  }

  }
